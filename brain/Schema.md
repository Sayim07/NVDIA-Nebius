## 3. File-Based Storage Schemas

### 3.1 Declarative Security Policy (`config/nemoclaw-policy.yaml`)
```yaml
policy_version: "1.0.0"
policy_name: "hermesshield-default-guard"
enforcement_mode: "strict"

filesystem:
  allowed_read_roots:
    - "data/workspace"
    - "data/skills"
    - "data/briefings"
    - "data/MEMORY.md"
  allowed_write_roots:
    - "data/workspace"
    - "data/briefings"
    - "data/skills"
  denied_path_patterns:
    - "^.*\\.env.*$"
    - "^.*id_rsa.*$"
    - "^.*\\.ssh.*$"
    - "^.*credentials.*$"
    - "^.*\\.git/.*$"

process:
  allowed_binaries:
    - "cat"
    - "ls"
    - "grep"
    - "python"
    - "pytest"
    - "echo"
  denied_binaries:
    - "curl"
    - "wget"
    - "nc"
    - "bash"
    - "sh"
    - "rm"
    - "dd"
  execution_timeout_seconds: 5.0

network:
  outbound_http_permitted: false
  allowed_destinations: []
```

## 4. Database Schema (SQLite: `data/hermesshield.db`)

HermesShield maintains four relational tables initialized at boot:

```sql
-- 1. Persistent Memories Table
CREATE TABLE IF NOT EXISTS memories (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    key TEXT UNIQUE NOT NULL,
    category TEXT NOT NULL,          -- 'user_preference', 'project_context', 'team'
    content TEXT NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tavily Cache Table (24-hour TTL)
CREATE TABLE IF NOT EXISTS tavily_cache (
    query_hash TEXT PRIMARY KEY,     -- SHA-256 of sanitized query string
    sanitized_query TEXT NOT NULL,
    search_response_json TEXT NOT NULL,
    extracted_markdown TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Security Audit Events Table
CREATE TABLE IF NOT EXISTS security_events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    command TEXT NOT NULL,
    status TEXT NOT NULL,            -- 'ALLOWED', 'BLOCKED'
    policy_rule TEXT NOT NULL,
    reason TEXT NOT NULL
);

-- 4. FinOps Telemetry Logs
CREATE TABLE IF NOT EXISTS telemetry_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    task_id TEXT NOT NULL,
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    tier_1_input_tokens INTEGER DEFAULT 0,
    tier_1_output_tokens INTEGER DEFAULT 0,
    tier_2_input_tokens INTEGER DEFAULT 0,
    tier_2_output_tokens INTEGER DEFAULT 0,
    tier_3_input_tokens INTEGER DEFAULT 0,
    tier_3_output_tokens INTEGER DEFAULT 0,
    total_cost_usd REAL NOT NULL,
    latency_ms REAL NOT NULL
);
```
